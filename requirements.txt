# --- Confucius4-R2T2 (https://github.com/netease-youdao/Confucius4-R2T2) ---

# Este repo (main.py, services/r2t2_stt.py) NO instala R2T2 aquí: corre como
# servidor WebSocket dentro de WSL2 (Linux real, GPU passthrough). Instalar
# ahí, no en este requirements.txt:
#   wsl
#   git clone https://github.com/netease-youdao/Confucius4-R2T2.git
#   cd Confucius4-R2T2 && python3 -m venv .venv && source .venv/bin/activate
#   pip install -e .
#   huggingface-cli download netease-youdao/Confucius4-R2T2 --local-dir ~/models/Confucius4-R2T2
#   ./run_start_server.sh start --model_path ~/models/Confucius4-R2T2
