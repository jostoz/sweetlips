# Pipecat con audio local + TTS Kokoro (offline, ONNX). Corre en Windows nativo.
# Pineado a 1.9.0: 1.11.0 tiene una regresión verificada donde el StartFrame
# nunca llega al final del pipeline incluso en un pipeline mínimo de un solo
# processor (PipelineTask/WorkerRunner se cuelgan en el timeout de setup).
pipecat-ai[kokoro]==1.9.0
# `pipecat-ai[kokoro]` instala `onnxruntime` (CPU). Probamos acelerar Kokoro
# TTS por GPU y ninguna opción funcionó en este Windows:
#   - onnxruntime-directml: crashea con "ConvTranspose" (ISTFTNet de Kokoro
#     no es compatible con ese backend) -> tira el TTS entero a "unusable"
#     tras 3 fallos.
#   - onnxruntime-gpu (CUDA): tira "cublasLt64_13.dll" faltante aunque
#     nvidia-cublas/nvidia-cudnn-cu13 estén instalados y el DLL cargue bien
#     standalone — onnxruntime no honra `os.add_dll_directory` para su
#     propio provider de CUDA en esta versión/plataforma.
# Quedamos en CPU (`onnxruntime~=1.24.3`, pineado a lo que pide pipecat):
# ~1-1.3s de síntesis por respuesta corta, estable, sin crashes.
onnxruntime~=1.24.3
